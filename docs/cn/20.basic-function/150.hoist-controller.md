
## 概述

本文档描述了空吊控制模块头文件`dji_psdk_hoist_controller.h`中的函数原型和数据类型定义。该模块用于控制和监测飞行器的空吊系统（含标准空吊、简易空吊和智能挂钩），开发者可以通过该模块获取空吊状态、挂钩状态、避障状态，以及发送收放绳、开关钩、脱困等控制指令。

::: tip 支持的机型说明
本模块仅适用于 FlyCart 100。各机型对功能模块的支持范围请参见 [开放功能列表](https://developer.dji.com/doc/payload-sdk-tutorial/cn/development-preparation/development-platform-overview/open-capability-list.html) 机型矩阵表。
:::

## 数据类型定义

### E_DjiHoistControllerCmdStatus

空吊控制指令。

```c
typedef enum {
    DJI_HOIST_CARGO_RELEASE_CMD     = 0,    释放货物
    DJI_HOIST_CARGO_STOP_CMD        = 1,    停止释放或收回货物
    DJI_HOIST_CARGO_RECEIVE_CMD     = 2,    收回货物
    DJI_HOIST_CARGO_MAINTENANCE_CMD = 4,    维护模式
    DJI_HOIST_CARGO_ESCAPE_TROUBLE  = 6,    脱困
} E_DjiHoistControllerCmdStatus;
```

### E_DjiHookControllerCmdStatus

挂钩控制指令。

```c
typedef enum {
    DJI_HOOK_CARGO_OLOSE_HOOK = 1,    关闭挂钩
    DJI_HOOK_CARGO_OPEN_HOOK  = 2,    打开挂钩
} E_DjiHookControllerCmdStatus;
```

### E_DjiHoistCableStatus

空吊缆绳状态。

```c
typedef enum {
    DJI_HOIST_CABLE_UP_DONE        = 0,       收绳完成
    DJI_HOIST_CABLE_TAKING_UP      = 0x1,     正在收绳
    DJI_HOIST_CABLE_PAUSE          = 0x2,     暂停动作
    DJI_HOIST_CABLE_TAKING_DOWN    = 0x3,     正在放绳
    DJI_HOIST_CABLE_DOWN_DONE      = 0x4,     放绳完成
    DJI_HOIST_CABLE_SYSTEM_PROTECT = 0xf,     系统保护/异常
    DJI_HOIST_CABLE_UNKONW         = 0xFF,    未知状态
} E_DjiHoistCableStatus;
```

### E_DjiHoistType

空吊类型。

```c
typedef enum {
    DJI_STANDARD_HOIST = 5,    标准空吊
    DJI_SIMPLE_HOIST   = 4,    简易空吊
    DJI_EEOR_LOAD      = 0,    错误/无空吊
} E_DjiHoistType;
```

### E_DjiSimpleHoistEscapeCmd

简易空吊脱困指令。

```c
typedef enum {
    DJI_ESCAPE_CTR_DISABLE     = 0,    禁用脱困
    DJI_ESCAPE_CTR_ENABLE      = 1,    启用脱困
    DJI_ROD_CTRL_CMD_OUT_ONLY  = 2,    仅推出（适用于杆式简易空吊）
    DJI_ROD_CTRL_CMD_IN_ONLY   = 3,    仅推入（适用于杆式简易空吊）
} E_DjiSimpleHoistEscapeCmd;
```

### psdk_receive_hoist_status_t

空吊系统状态。

```c
typedef struct {
    E_DjiHoistCableStatus cable_state;    当前缆绳状态
    uint8_t  can_status;                  bit0: 可放绳; bit1: 可收绳; bit7: 可控制空吊
    int32_t  goods_weight;                货物重量，单位：g
    uint32_t max_goods_weight;            最大货物重量，单位：g
    uint16_t cargo_swing_speed;           货物摆动速度
    uint16_t cur_cable_length;            当前缆绳长度，单位：m
    uint16_t cable_force;                 缆绳拉力，单位：g
    uint16_t cable_angle;                 缆绳角度，单位：0.1 度
    uint8_t  is_reach_top;                货物是否到达顶部，1: 是; 0: 否
    uint8_t  is_reach_max_round;          缆绳是否达到最大圈数
    uint8_t  cable_direction_check_state; 缆绳检查状态：0x00: 空闲; 0x01: 检查中; 0x02: 检查结果正确; 0x03: 检查结果错误
    uint16_t cable_speed;                 当前缆绳动作速度，单位：cm/s
} psdk_receive_hoist_status_t;
```

### psdk_receive_hook_status_t

智能挂钩状态。

```c
typedef struct {
    uint8_t  version;          版本号
    uint8_t  steer_state;      挂钩状态：0x0: 未知; 0x1: 关闭; 0x2: 打开; 0x3: 中间位置
    int16_t  cargo_weight;     货物重量，单位：0.1 kg
    uint16_t cargo_warnig;     挂钩警告：0 为正常; bit0: 电池未就位; bit1: 电池低电; bit2: 电池低压; bit3: 舵机未就位; bit4: 舵机堵转; bit5: 断连; bit14: 舵机异常; bit15: 挂钩未激活
    uint16_t cargo_error;      挂钩故障，0 为正常
    uint8_t  ctrl_state;       挂钩控制状态：0: 正常; 1: 舵机上电中; 2: 舵机已上电; 3: 舵机移动中(开); 4: 舵机正常停止; 5: 舵机移动中(关)
} psdk_receive_hook_status_t;
```

### psdk_receive_simple_hoist_status_t

简易空吊状态。

```c
typedef struct {
    uint8_t  version;          版本号
    uint8_t  sensor_damage;    传感器状态：0x0: 正常; 0x1: 损坏
    uint8_t  weight_k_err;     称重校准错误：0x0: 正常; 其他: 错误
    int16_t  force_kg;         力值，force_kg / 100 = kg
    int16_t  angle_deg;        角度值，angle_deg / 100 = 度
} psdk_receive_simple_hoist_status_t;
```

### psdk_receive_avoid_status_t

飞行避障状态。

```c
typedef struct {
    uint8_t is_cargo_avoid:1;        货物避障中
    uint8_t is_avoid_stop_finish:1;  避障飞行已停止
    uint8_t is_start_avoid:1;        避障已启动，开始减速
    uint8_t reserve:5;               保留
} psdk_receive_avoid_status_t;
```
