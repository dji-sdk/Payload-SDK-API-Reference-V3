```c
typedef enum {
    /*!拍照模式。在本模式下，用户可以拍摄照片。
     */
    DJI_CAMERA_MANAGER_WORK_MODE_SHOOT_PHOTO = 0,
    /*!录像模式。在本模式下，用户可以录制视频。
     */
    DJI_CAMERA_MANAGER_WORK_MODE_RECORD_VIDEO = 1,
    /*!回放模式。在本模式下，用户可以回看照片和视频并且可以删除文件。被Phantom 3专业相机、X3、X5、X5R相机以及Phantom 4相机支持。不被Z30、X5S、X4S、Phantom 4 Pro、Mavic Pro、Phantom 3 Standard、Phantom 3 Advanced、Phantom 3 4K以及 Osmo series支持。
     */
    DJI_CAMERA_MANAGER_WORK_MODE_PLAYBACK = 2,
    /*!在本模式下，用户可以在移动设备下载媒体文件。在 X5 相机以及 X5R 相机上不支持。 
    */
    DJI_CAMERA_MANAGER_WORK_MODE_MEDIA_DOWNLOAD = 3,
    /*!在本模式下，直播分辨率和帧率为1080i50 (PAL)或者720p60 (NTSC)。在这个模式中可以录制视频。录像时也可以拍照。从公告模式中国退出的唯一方式就是切换模式为RECORD_VIDEO。仅在Inspire 2中支持。
     */
     DJI_CAMERA_MANAGER_WORK_MODE_BROADCAST = 4,
     /*!本模式为未知模式。
     */
    DJI_CAMERA_MANAGER_WORK_MODE_WORK_MODE_UNKNOWN = 0xFF,
} E_DjiCameraManagerWorkMode;
```

```c
typedef enum {
    /*!设置相机拍摄单张照片。
     */
    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_SINGLE = 0x01,
    /*!设置相机拍摄一张 HDR 照片。X5 相机、X5R 相机、XT 相机、Z30 相机、Phantom 4 Pro 相机、X4S 相机和 X5S 相机不支持HDR模式。

    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_HDR = 0x02,

     */
    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_BURST = 0x04,
    /*!自动包围曝光拍摄（AEB）。在本模式下，您可以在不同曝光下快速拍摄多张照片（默认为 3 张），而无需手动更改帧之间的任何设置。XT 相机和Z30 相机不支持AEB模式.
     */
    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_AEB = 0x05,
    /*!设置相机以设定的时间间隔连续拍摄一张（或多张照片）。任何质量的JPEG 格式相片的最小间隔为 2 秒。对于除 X4S、X5S 和 Phantom 4 Pro 相机以外的所有相机：RAW 或 RAW+JPEG 格式的最小间隔为 10 秒。 对于 X4S、X5S 和 Phantom 4 Pro 相机，RAW 或 RAW+JPEG dformat 的最小间隔为 5 秒。
     */
    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_INTERVAL = 0x06,
    /*!设置相机连续拍摄 RAW 照片。使用 getRAWPhotoBurstCount 检查已拍摄的照片数量。 仅 X5S 支持本模式。

    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_RAW_BURST = 0x09,

     */
    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_REGIONAL_SR = 0x16,
    /*!本模式为未知模式。
     */
    DJI_CAMERA_MANAGER_SHOOT_PHOTO_MODE_UNKNOWN = 0xFF,
} E_DjiCameraManagerShootPhotoMode;
```
